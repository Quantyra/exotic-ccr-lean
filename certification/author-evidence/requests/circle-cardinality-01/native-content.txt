theorem circleUnitary_injective : Function.Injective (fun z : Circle => (z : C)) := by
  intro a b h
  have h₁ : (a : C) = (b : C) := by simpa using h
  have h₂ : a = b := by
    have h₃ : (a : C) = (b : C) := h₁
    have h₄ : a = b := by
      apply Circle.coe_injective
      simpa using h₃
    exact h₄
  simpa using h₂
